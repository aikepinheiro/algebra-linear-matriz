Saída do programa em C
Passe o mouse ou toque nas células das matrizes acima para ver a correspondência de posições [l][c] → [c][l].
main.c — Código Fonte Original 
#include <stdio.h>
#include <stdbool.h>

int main()
{
    int n = 0;
    int l, c;
    
    printf("Insira a ordem n da matriz:");
    scanf("%d", &n);
    
    int Ma[n][n];
    int MaT[n][n];
    
    for(l = 0; l<n; l++){
        for(c = 0; c<n; c++){
            printf("Insira a posição [%d][%d]:", l, c);
            scanf("%d", &Ma[l][c]);
        }
    
    }
    
    printf("\nMatriz Principal:\n");
    
    for(l = 0; l<n; l++){
        for(c = 0; c<n; c++){
            printf("%4d", Ma[l][c]);
        }
    printf("\n");
    }
    
    for(l = 0; l<n; l++){
        for(c = 0; c<n; c++){
            MaT[l][c] = Ma[c][l];
    
        }
    }
    
    printf("\nMatriz Transposta:\n");
    
    for(l = 0; l<n; l++){
        for(c = 0; c<n; c++){
            printf("%4d", MaT[l][c]);
        } 
        printf("\n");
    }   
    
    bool simetrica = true;
    
    for(l = 0; l < n; l++){
        for(c = 0; c < n; c++){
            if(Ma[l][c] != MaT[l][c]){
                simetrica = false;
                break;
            }
        }
        if(!simetrica){
            break;
        }
    }
    
    if(simetrica == true) {
        printf("\nA matriz é simetrica.\n");
    } else {
        printf("\nA matriz não é simetrica.\n");
    }
    
    return 0;
}
